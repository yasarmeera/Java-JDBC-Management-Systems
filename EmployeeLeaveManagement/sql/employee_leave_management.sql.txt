-- Employee Leave Management System


-- Create database
CREATE DATABASE IF NOT EXISTS employee_leave_system;
USE employee_leave_system;


-- Employees table
CREATE TABLE employees (
    employee_id INT PRIMARY KEY,
    employee_name VARCHAR(100) NOT NULL,
    department VARCHAR(50),
    designation VARCHAR(50));


-- Leave Balance table
CREATE TABLE leave_balance (
    balance_id INT PRIMARY KEY,
    employee_id INT NOT NULL,
    total_leave INT DEFAULT 12,
    used_leave INT DEFAULT 0,
    remaining_leave INT DEFAULT 12,

    FOREIGN KEY (employee_id)
        REFERENCES employees(employee_id));


-- Leave Requests table
CREATE TABLE leave_requests (
    leave_id INT PRIMARY KEY,
    employee_id INT NOT NULL,
    leave_type VARCHAR(50) NOT NULL,
    from_date DATE NOT NULL,
    to_date DATE NOT NULL,
    status VARCHAR(20) DEFAULT 'Pending',

    FOREIGN KEY (employee_id)
        REFERENCES employees(employee_id));

-- Join: Leave request details
SELECT
    lr.leave_id,
    e.employee_id,
    e.employee_name,
    e.department,
    lr.leave_type,
    lr.from_date,
    lr.to_date,
    lr.status
FROM leave_requests lr
JOIN employees e
    ON lr.employee_id = e.employee_id;


-- Join: Employee leave balance
SELECT
    e.employee_id,
    e.employee_name,
    lb.total_leave,
    lb.used_leave,
    lb.remaining_leave
FROM employees e
JOIN leave_balance lb
    ON e.employee_id = lb.employee_id;
